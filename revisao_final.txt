function mensagemErro(error: unknown): string {
  const resposta = (
    error as {
      response?: {
        data?: {
          detail?: string | { msg?: string }[];
        };
      };
    }
  )?.response?.data?.detail;

  if (typeof resposta === "string") {
    return resposta;
  }

  if (Array.isArray(resposta)) {
    return resposta.map((item) => item.msg || "Dados invÃ¡lidos").join("\n");
  }

  if ((error as { response?: unknown })?.response) {
    return "NÃ£o foi possÃ­vel concluir a operaÃ§Ã£o. Verifique os dados e tente novamente.";
  }

  return "NÃ£o foi possÃ­vel conectar ao servidor.";
}

function formatarData(data: string | null): string {
  if (!data) return "Nunca acessou";

  return new Date(data).toLocaleString("pt-BR", {
    dateStyle: "short",
    timeStyle: "short",
  });
}
